- [ ] **Step 2: Add the `loadTree` method**

This fetches epics from active projects, then fetches children for each epic in parallel batches of 5. Builds a tree JSON: `{ name: "Work Map", children: [{ name: projectKey, children: [{ name: epicKey, ... }] }] }`.

```swift
func loadTree(appState: AppState) async {
    guard appState.isJiraConfigured else {
        withAnimation(.none) { error = "Jira not configured." }
        return
    }
    withAnimation(.none) { isLoading = true; error = nil }

    let projectKeys = appState.activeJiraProjectKeys
    guard !projectKeys.isEmpty else {
        withAnimation(.none) { isLoading = false; error = "No active Jira projects." }
        return
    }

    do {
        // Fetch epics across all active projects
        let quotedKeys = projectKeys.map { k in
            // Quote JQL reserved words
            let reserved = ["DO", "IF", "OR", "IN", "IS", "ON", "TO", "AS", "BY", "OF", "NO", "IT", "GO", "AT"]
            return reserved.contains(k.uppercased()) ? "\"\(k)\"" : k
        }.joined(separator: ", ")
        let epicJQL = "issuetype = Epic AND project IN (\(quotedKeys)) ORDER BY project ASC, key ASC"

        let epicResult = try await jiraService.searchIssues(
            baseURL: appState.jiraBaseURL, email: appState.jiraEmail,
            apiToken: appState.jiraAPIToken, jql: epicJQL,
            fields: ["summary", "status", "priority", "assignee", "updated",
                     appState.storyPointsFieldId],
            maxResults: 500
        )

        // Fetch children for each epic in parallel (batches of 5)
        var epicChildMap: [String: [JiraIssue]] = [:]
        let epicKeys = epicResult.issues.map(\.key)

        await withTaskGroup(of: (String, [JiraIssue]).self) { group in
            for epicKey in epicKeys {
                group.addTask {
                    let childJQL = "parent = \(epicKey) ORDER BY status ASC, key ASC"
                    let children = (try? await self.jiraService.searchIssues(
                        baseURL: appState.jiraBaseURL, email: appState.jiraEmail,
                        apiToken: appState.jiraAPIToken, jql: childJQL,
                        fields: ["summary", "status", "priority", "issuetype",
                                 "assignee", "updated", appState.storyPointsFieldId],
                        maxResults: 200
                    ))?.issues ?? []
                    return (epicKey, children)
                }
            }
            for await (epicKey, children) in group {
                epicChildMap[epicKey] = children
            }
        }

        // Build tree JSON
        let tree = buildTreeJSON(epics: epicResult.issues, childMap: epicChildMap)
        let jsonData = try JSONSerialization.data(withJSONObject: tree, options: [])
        let jsonString = String(data: jsonData, encoding: .utf8) ?? "{}"

        // Compute stats
        let allIssues = epicResult.issues.count + epicChildMap.values.reduce(0, { $0 + $1.count })
        let doneCount = epicResult.issues.filter { $0.fields.status?.statusCategory?.key == "done" }.count
            + epicChildMap.values.flatMap({ $0 }).filter { $0.fields.status?.statusCategory?.key == "done" }.count
        let pct = allIssues > 0 ? Double(doneCount) / Double(allIssues) * 100 : 0

        withAnimation(.none) {
            treeJSON = jsonString
            epicCount = epicResult.issues.count
            issueCount = allIssues
            completionPct = pct
            isLoading = false
        }
        Self.log.notice("loadTree: \(epicResult.issues.count, privacy: .public) epics, \(allIssues, privacy: .public) total issues")
    } catch {
        withAnimation(.none) {
            self.error = error.localizedDescription
            isLoading = false
        }
    }
}
```

- [ ] **Step 3: Add the `buildTreeJSON` helper**

Converts the flat epic + children data into a nested dictionary tree grouped by project.

```swift
private func buildTreeJSON(epics: [JiraIssue], childMap: [String: [JiraIssue]]) -> [String: Any] {
    // Group epics by project key
    var projectGroups: [String: [JiraIssue]] = [:]
    for epic in epics {
        let projectKey = String(epic.key.split(separator: "-").first ?? "?")
        projectGroups[projectKey, default: []].append(epic)
    }

    let projectNodes: [[String: Any]] = projectGroups.sorted(by: { $0.key < $1.key }).map { (projectKey, projectEpics) in
        let epicNodes: [[String: Any]] = projectEpics.map { epic in
            let children = childMap[epic.key] ?? []
            let childNodes: [[String: Any]] = children.map { child in
                [
                    "key": child.key,
                    "name": child.fields.summary ?? child.key,
                    "type": child.fields.issuetype?.name?.lowercased() ?? "task",
                    "status": child.fields.status?.name ?? "Unknown",
                    "statusCategory": child.fields.status?.statusCategory?.key ?? "undefined",
                    "sp": child.fields.storyPoints as Any,
                    "assignee": child.fields.assignee?.displayName ?? "Unassigned",
                    "updated": child.fields.updated ?? ""
                ]
            }
            return [
                "key": epic.key,
                "name": epic.fields.summary ?? epic.key,
                "type": "epic",
                "status": epic.fields.status?.name ?? "Unknown",
                "statusCategory": epic.fields.status?.statusCategory?.key ?? "undefined",
                "sp": epic.fields.storyPoints as Any,
                "assignee": epic.fields.assignee?.displayName ?? "Unassigned",
                "updated": epic.fields.updated ?? "",
                "children": childNodes
            ]
        }
        return [
            "key": projectKey,
            "name": projectKey,
            "type": "project",
            "status": "",
            "statusCategory": "",
            "children": epicNodes
        ]
    }

    return [
        "name": "Work Map",
        "key": "root",
        "type": "root",
        "status": "",
        "statusCategory": "",
        "children": projectNodes
    ]
}
```

```swift
import SwiftUI

struct WorkMapView: View {
    @EnvironmentObject var appState: AppState
    @State private var vm = WorkMapViewModel()

    var body: some View {
        VStack(spacing: 0) {
            topBar
            Divider()
            ZStack(alignment: .bottomLeading) {

            Task { await vm.loadTree(appState: appState) }
```

- [ ] **Step 2: Create WorkMapWebView (NSViewRepresentable)**

```swift
struct WorkMapWebView: NSViewRepresentable {
    let treeJSON: String
    let statusFilter: String
    let searchText: String
    let onNodeClick: (String) -> Void

    func makeNSView(context: Context) -> WKWebView {
        let config = WKWebViewConfiguration()
        config.userContentController.add(context.coordinator, name: "nodeClick")
        let wv = WKWebView(frame: .zero, configuration: config)
        wv.setValue(false, forKey: "drawsBackground")
        context.coordinator.webView = wv

        // Load bundled HTML
        if let htmlURL = Bundle.main.url(forResource: "work_map", withExtension: "html") {
            wv.loadFileURL(htmlURL, allowingReadAccessTo: htmlURL.deletingLastPathComponent())
        }
        return wv
    }

    func updateNSView(_ wv: WKWebView, context: Context) {
        let coord = context.coordinator
        // Push data when treeJSON changes
        if coord.lastTreeJSON != treeJSON, !treeJSON.isEmpty {
            coord.lastTreeJSON = treeJSON
            let escaped = treeJSON.replacingOccurrences(of: "\\", with: "\\\\")
                .replacingOccurrences(of: "'", with: "\\'")
            wv.evaluateJavaScript("if(window.loadData) window.loadData('\(escaped)')") { _, _ in }
        }
        // Push filter changes
        if coord.lastStatusFilter != statusFilter {
            coord.lastStatusFilter = statusFilter
            wv.evaluateJavaScript("if(window.filterByStatus) window.filterByStatus('\(statusFilter)')") { _, _ in }
        }
        // Push search changes
        if coord.lastSearchText != searchText {
            coord.lastSearchText = searchText
            let escapedSearch = searchText.replacingOccurrences(of: "'", with: "\\'")
            wv.evaluateJavaScript("if(window.searchNodes) window.searchNodes('\(escapedSearch)')") { _, _ in }
        }
    }

    func makeCoordinator() -> Coordinator { Coordinator(onNodeClick: onNodeClick) }

    class Coordinator: NSObject, WKScriptMessageHandler {
        weak var webView: WKWebView?
        var lastTreeJSON = ""
        var lastStatusFilter = ""
        var lastSearchText = ""
        let onNodeClick: (String) -> Void

        init(onNodeClick: @escaping (String) -> Void) {
            self.onNodeClick = onNodeClick
        }

        func userContentController(_ controller: WKUserContentController,
                                   didReceive message: WKScriptMessage) {
            if message.name == "nodeClick", let key = message.body as? String {
                Task { @MainActor in onNodeClick(key) }
            }
        }
    }
}
```

```swift
struct MyWorkPanel: View {
    @EnvironmentObject var appState: AppState
    @State private var selectedTab = 0

    private static let tabMap: [String: Int] = [
        "jira_todo": 0, "jira_filters": 1, "jira_boards": 2, "work_map": 3
    ]
    private static let tabLabels = ["Tickets", "Filters", "Boards", "Work Map"]

    var body: some View {
        VStack(spacing: 0) {
            Picker("", selection: $selectedTab) {
                Text("Tickets").tag(0)
                Text("Filters").tag(1)
                Text("Boards").tag(2)
                Text("Work Map").tag(3)
            }
            .pickerStyle(.segmented)
            .padding(.horizontal, 16).padding(.vertical, 8)

            Divider()

            Group {
                switch selectedTab {
                case 0: TodoDashboardView()
                case 1: SavedFiltersView()
                case 2: BoardsView()
                case 3: WorkMapView()
                default: EmptyView()
                }
            }
            .frame(maxWidth: .infinity, maxHeight: .infinity)
        }
        .onAppear { consumePendingTab(); updateSubTab() }
        .onChange(of: appState.pendingTabId) { consumePendingTab() }
        .onChange(of: selectedTab) { updateSubTab() }
    }

    private func consumePendingTab() {
        if let id = appState.pendingTabId, let tab = Self.tabMap[id] {
            selectedTab = tab
            appState.pendingTabId = nil
        }
    }

    private func updateSubTab() {
        let key = Self.tabMap.first(where: { $0.value == selectedTab })?.key
        appState.currentSubTab = key ?? Self.tabLabels[selectedTab]
    }
}
```
